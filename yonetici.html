function restoreTimestamp(v){
 if(v&&typeof v==="object"&&!Array.isArray(v)){
  if(typeof v.__timestamp==="string"){const ms=Date.parse(v.__timestamp);if(!Number.isFinite(ms))throw Error("Geçersiz zaman damgası.");return Timestamp.fromMillis(ms)}
  if(typeof v.seconds==="number"&&typeof v.nanoseconds==="number"&&Object.keys(v).every(k=>["seconds","nanoseconds"].includes(k)))return new Timestamp(v.seconds,v.nanoseconds);
  return Object.fromEntries(Object.entries(v).map(([k,x])=>[k,restoreTimestamp(x)]));
 }
 if(Array.isArray(v))return v.map(restoreTimestamp);
 return v;
}
